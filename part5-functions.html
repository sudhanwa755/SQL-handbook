5.1

Understanding NULL

NULL means "no value / unknown" — it is NOT zero, NOT an empty string, NOT false. It is the absence of any value.

NULL in Arithmetic

Any arithmetic with NULL produces NULL:

SQL — NULL Arithmetic
-- Salary + Commission:  NULL for employees without commission
SELECT EmpName, Salary, Commission,
       Salary + Commission AS TotalPay
FROM   EMPLOYEES;

5.2

NVL — Replace NULL with a Default Value

When to use: To substitute a replacement value whenever a column is NULL. Available in Oracle-style SQL and HSQLDB.

Syntax: NVL(column, replacement_value)

SQL — NVL Function
-- Treat NULL commission as 0 before adding
SELECT
    EmpName,
    Salary,
    NVL(Commission, 0) AS Commission,
    Salary + NVL(Commission, 0) AS TotalPay
FROM   EMPLOYEES;
Result: NVL replaces NULL with 0
EmpNameSalaryCommissionTotalPay
Alice Johnson85000500090000
Bob Smith55000055000
Carol Wang920008000100000
David Kumar78000078000
Eva Martinez62000350065500
5.3

COALESCE — Return First Non-NULL Value

When to use: When you have multiple columns and want the first one that is not NULL. More flexible than NVL (works with multiple arguments).

Syntax: COALESCE(val1, val2, val3, ...) — returns the first non-NULL value.

SQL — COALESCE
-- Try Commission first, then Bonus, then default 0
SELECT
    EmpName,
    COALESCE(Commission, Bonus, 0) AS ExtraPay
FROM   EMPLOYEES;

-- Replace NULL with a descriptive string
SELECT
    EmpName,
    COALESCE(CAST(Commission AS VARCHAR(20)), 'No Commission') AS CommStatus
FROM   EMPLOYEES;

5.4

CAST — Type Conversion

When to use: To explicitly convert a value from one data type to another.

Syntax: CAST(expression AS target_type)

SQL — CAST Examples
-- Convert salary (INT) to DECIMAL for precise math
SELECT CAST(Salary AS DECIMAL(10,2)) AS SalaryDecimal
FROM   EMPLOYEES;

-- Convert number to string (for CONCAT)
SELECT
    EmpName || ' earns ' || CAST(Salary AS VARCHAR(10)) AS Summary
FROM   EMPLOYEES;

-- Convert string date to DATE type
SELECT CAST('2024-06-15' AS DATE) AS ParsedDate FROM DUAL;

-- Convert to integer (truncates decimal)
SELECT CAST(3.99 AS INT) AS TruncatedNum FROM DUAL; -- Result: 3
Result: EmpName || ' earns ' || CAST(Salary...)

5.5

String Functions

CONCAT (||) — Joining Strings

SQL — String Concatenation
-- Using CONCAT function
SELECT CONCAT(EmpName, ' — Dept: ', Dept) AS EmployeeInfo
FROM   EMPLOYEES;

-- Using || operator (HSQLDB)
SELECT EmpName || ' (' || Dept || ')' AS EmployeeInfo
FROM   EMPLOYEES;
Result: CONCAT output
EmployeeInfo
Alice Johnson (IT)
Bob Smith (HR)
Carol Wang (IT)

Case Functions

SQL — UPPER, LOWER, INITCAP
SELECT
    EmpName,
    UPPER(EmpName)    AS UpperName,    -- ALICE JOHNSON
    LOWER(EmpName)    AS LowerName,    -- alice johnson
    UPPER(Dept)       AS DeptUpper
FROM   EMPLOYEES;

-- Case-insensitive search using UPPER
SELECT * FROM EMPLOYEES
WHERE UPPER(Dept) = UPPER('it');

Length & Trimming

SQL — LENGTH, TRIM, LTRIM, RTRIM
SELECT
    EmpName,
    LENGTH(EmpName)          AS NameLength,
    TRIM('   Alice   ')       AS Trimmed,      -- 'Alice'
    LTRIM('   Alice')         AS LeftTrimmed,  -- 'Alice'
    RTRIM('Alice   ')         AS RightTrimmed  -- 'Alice'
FROM   EMPLOYEES;

Substring & Position

SQL — SUBSTRING, POSITION, REPLACE
SELECT
    EmpName,
    -- Get first 5 characters of the name
    SUBSTRING(EmpName, 1, 5)       AS ShortName,
    -- Find position of space (to split first/last name)
    POSITION(' ' IN EmpName)       AS SpacePos,
    -- Replace 'Johnson' with 'J.'
    REPLACE(EmpName, 'Johnson', 'J.') AS AbbrevName
FROM   EMPLOYEES;

Date Functions

TO_CHAR — Format a Date or Number as a String

SQL — TO_CHAR with Date
-- Format date in various ways
SELECT
    EmpName,
    JoinDate,
    TO_CHAR(JoinDate, 'DD-MON-YYYY')   AS FormattedDate,   -- 15-MAR-2020
    TO_CHAR(JoinDate, 'Month YYYY')      AS MonthYear,        -- March 2020
    TO_CHAR(JoinDate, 'YYYY')             AS JoinYear           -- 2020
FROM   EMPLOYEES;

-- Format a number as string with TO_CHAR
SELECT TO_CHAR(Salary, '999,999') AS FormattedSalary
FROM   EMPLOYEES;

TO_DATE — Parse a String as a Date

SQL — TO_DATE
-- Convert string to DATE (for comparison or storage)
SELECT * FROM EMPLOYEES
WHERE  JoinDate >= TO_DATE('01-JAN-2020', 'DD-MON-YYYY');

-- Calculate days since joining
SELECT
    EmpName,
    JoinDate,
    CURRENT_DATE - JoinDate AS DaysSinceJoining
FROM   EMPLOYEES;

Other Useful Date Functions

SQL — Date Utility Functions
SELECT
    CURRENT_DATE                          AS Today,
    CURRENT_TIMESTAMP                     AS NowDateTime,
    EXTRACT(YEAR  FROM CURRENT_DATE)  AS CurrentYear,
    EXTRACT(MONTH FROM CURRENT_DATE)  AS CurrentMonth,
    EXTRACT(DAY   FROM CURRENT_DATE)  AS CurrentDay
FROM   DUAL;

Math Functions

SQL — Math Functions
SELECT
    ROUND(85750.678, 2)   AS Rounded,      -- 85750.68
    FLOOR(85750.9)        AS FloorVal,     -- 85750 (round DOWN)
    CEILING(85750.1)      AS CeilVal,      -- 85751 (round UP)
    ABS(-5000)            AS AbsVal,       -- 5000 (absolute value)
    MOD(17, 5)            AS Remainder,    -- 2 (17 mod 5)
    POWER(2, 10)          AS TwoToTen,     -- 1024
    SQRT(144)             AS SqrtVal,      -- 12.0
    SIGN(-85)              AS SignVal        -- -1 (negative number)
FROM   DUAL;

🔑 Quick Function Reference
  • String: UPPER, LOWER, LENGTH, TRIM, LTRIM, RTRIM, SUBSTRING, POSITION, REPLACE, CONCAT (||)
  • NULL: NVL(col, default), COALESCE(v1, v2, ...)
  • Date: CURRENT_DATE, CURRENT_TIMESTAMP, EXTRACT, TO_CHAR, TO_DATE
  • Math: ROUND, FLOOR, CEILING, ABS, MOD, POWER, SQRT, SIGN
  • Type: CAST(value AS type)
